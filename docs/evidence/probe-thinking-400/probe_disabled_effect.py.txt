"""验证 thinking=disabled 真的压制 reasoning_content(两连轮,含工具回传形状)。"""
import json

from dotenv import dotenv_values
from openai import OpenAI

env = dotenv_values(".env")
client = OpenAI(
    base_url=env["PATCHPILOT_LLM_BASE_URL"], api_key=env["PATCHPILOT_LLM_API_KEY"],
    timeout=120, max_retries=0,
)
MODEL = env["PATCHPILOT_LLM_MODEL"]
TOOLS = [{"type": "function", "function": {
    "name": "finish", "description": "结束",
    "parameters": {"type": "object", "properties": {"success": {"type": "boolean"}, "summary": {"type": "string"}},
                   "required": ["success", "summary"]}}}]

msgs = [{"role": "system", "content": "You are a code agent."},
        {"role": "user", "content": "查看 src/ 下有哪些文件,然后 finish。"}]
for i in range(2):
    r = client.chat.completions.create(
        model=MODEL, messages=msgs, tools=TOOLS, max_tokens=200,
        extra_body={"thinking": {"type": "disabled"}},
    )
    m = r.choices[0].message
    rc = getattr(m, "reasoning_content", None)
    print(f"turn{i+1}: finish={r.choices[0].finish_reason} reasoning_content={'None' if not rc else f'len={len(rc)}'} tool_calls={len(m.tool_calls or [])}")
    msgs.append({"role": "assistant", "content": m.content or "",
                 **({"reasoning_content": rc} if rc else {}),
                 **({"tool_calls": [{"type": "function", "id": t.id, "function": {"name": t.function.name, "arguments": t.function.arguments}} for t in m.tool_calls]} if m.tool_calls else {})})
    if m.tool_calls:
        for t in m.tool_calls:
            msgs.append({"role": "tool", "tool_call_id": t.id, "content": "src/main.py\nsrc/util.py"})
print("final history roles:", [x["role"] for x in msgs])
